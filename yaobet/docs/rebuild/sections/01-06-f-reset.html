Global import order (globals.css)

src/app/globals.css is the only global sheet, and it is imported once by layout.tsx (import "./globals.css"). It @imports, in this order: ../styles/reset.css → tokens.css → themes.css → brands.css → typography.css → motion.css. Its own rules follow.

Verbatim src/app/globals.css
/* ============================================================================
 * Yaobet — globals.css : the ONLY global sheet (SPEC-04 PART 2)
 * ----------------------------------------------------------------------------
 * Imports the design-token layers in order, sets the dark canvas + base type,
 * and wires the reduced-motion contract. Components ship their own scoped
 * *.module.css and read var(--…) from the layers below — they never re-author
 * tokens and never add to this global sheet.
 * Import order matters: reset → tokens (defines vars) → themes/brands
 * (re-point) → typography/motion (consume vars).
 * ==========================================================================*/

@import "../styles/reset.css";
@import "../styles/tokens.css";
@import "../styles/themes.css";
@import "../styles/brands.css";
@import "../styles/typography.css";
@import "../styles/motion.css";

html {
  height: 100%;
  background: var(--shell);
  color-scheme: dark;
  /* No horizontal page scroll, ever (SPEC-04 8.1). Wide content scrolls in
   * its own Scroller, not the body. */
  overflow-x: hidden;
}

body {
  min-height: 100%;
  background: var(--canvas);
  color: var(--text-hi);
  font-family: var(--brand-font-body);
  font-size: var(--t-body);
  line-height: var(--lh-body);
  overflow-x: hidden;
}

/* Headings inherit the body face; sizes are set per-component via --t-* tokens.
 * Provide a sane default so unstyled headings are not browser-huge. */
h1,
h2,
h3,
h4,
h5,
h6 {
  font-weight: var(--weight-bold);
  line-height: var(--lh-tight);
}

/* Respect the AA muted floor for any raw secondary text. */
small {
  font-size: var(--t-sm);
  color: var(--text-mid);
}

Verbatim src/app/loading.tsx
/* ============================================================================
 * Yaobet — HOME route loading UI (App Router Suspense fallback).
 * ----------------------------------------------------------------------------
 * Shown while page.tsx's server fetches (jackpot/leaderboard/games/…) are in
 * flight. Per the §F dynamic-data contract this is the MANDATORY loading state:
 * branded Skeletons in each band's reserved footprint — NEVER a placeholder
 * digit, a "0", or a last-known number dressed as live. Reduced-motion → the
 * Skeleton sweep falls back to a static tint via motion.css.
 * ==========================================================================*/

import { Skeleton } from "@/ui";
import styles from "@/components/home/Home.module.css";

export default function HomeLoading() {
  return (
    <div className={styles.page} aria-busy="true" aria-label="Đang tải trang chủ">
      <div className={styles.main}>
        {/* hero */}
        <Skeleton shape="block" width="100%" height={200} />
        {/* category launcher row */}
        <div className={styles.skelRow}>
          {Array.from({ length: 5 }).map((_, i) => (
            <Skeleton key={i} shape="pill" width={64} height={64} />
          ))}
        </div>
        {/* live band */}
        <Skeleton shape="block" width="100%" height={140} />
        {/* game grid */}
        <Skeleton shape="text" width="40%" height="1.6em" />
        <div className={styles.skelGrid}>
          {Array.from({ length: 6 }).map((_, i) => (
            <Skeleton key={i} shape="block" width="100%" height={120} />
          ))}
        </div>
        {/* vip aisle */}
        <Skeleton shape="block" width="100%" height={160} />
      </div>
    </div>
  );
}
