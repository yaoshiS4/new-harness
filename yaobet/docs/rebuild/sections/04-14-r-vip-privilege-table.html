PrivilegeTable (/tier)

1. Purpose

Column-per-tier matrix of rebate rate and perks with a frozen label column.

2. Source

src/components/vip/PrivilegeTable.tsx, PrivilegeTable.module.css

3. Structure

Scroller(freezeFirstColumn snap=false peek=false label="Bảng quyền lợi").scroller
  div.col.labelCol
    div.cell.headCell  "Hạng"
    div.cell.rowLabel  "Hoàn trả"
    per perkRow: div.cell.rowLabel  perk
  per tier: div.col[.currentCol]
    div.cell.headCell.tierHead: span.tierName name · span.tierLevel "Hạng {n}"
    div.cell.rateCell  "Hoàn trả {x}%"
    per perkRow: div.cell → Icon check 18 .yes label="Có"  |  span.no aria-label="Không" "—"

perkRows = union of all tier perks, in order: Hoàn trả cơ bản, Hoàn trả 0,8%, Quà sinh nhật, Hoàn trả 1,0%, Rút nhanh, Hoàn trả 1,2%, Quản lý riêng, Hoàn trả 1,5%, Sự kiện VIP (9 rows). A tier gets ✓ only on its own two perks (non-cumulative).

/* PrivilegeTable — column-per-tier matrix; first column pinned by Scroller
 * freezeFirstColumn. Tokens only. */

.scroller {
  border-radius: var(--radius-card);
  border: 1px solid var(--border);
  overflow: hidden;
}

/* each column is a vertical stack of cells; rows align by equal cell heights.
 * @375 the columns keep their min-width and the Scroller scrolls horizontally
 * (frozen first column). On desktop the table fills the 1424 container instead
 * of leaving dead space to its right — see the min-width:768px block below. */
/* Every column reserves the same 1px side rules and the same min-width, so the
 * current-tier column is marked by COLOUR alone and all tier columns stay
 * geometrically identical (ruling B1: no size-ranking by tier). */
.col {
  display: flex;
  flex-direction: column;
  flex: 0 0 auto;
  min-width: 132px;
  border-left: 1px solid transparent;
  border-right: 1px solid transparent;
}
.labelCol {
  min-width: 148px;
}

/* Desktop: stretch the tier columns to share the full container width so the
 * table has no empty area to its right. The label column keeps a fixed basis;
 * the tier columns grow equally (flex: 1) past their mobile min-width. The rail
 * (Scroller) is forced to 100% so there IS leftover space to distribute. */
@media (min-width: 768px) {
  /* the Scroller renders <outer><rail/></outer>; .scroller is on outer, so the
   * direct child div is the scroll rail. Force it full-width so flex-grow has
   * room to distribute across the tier columns. */
  .scroller > div {
    width: 100%;
  }
  .labelCol {
    flex: 0 0 auto;
    min-width: 180px;
  }
  .col:not(.labelCol) {
    flex: 1 1 0;
    min-width: 0;
  }
}
.currentCol {
  background: var(--tint);

  border-left-color: var(--brand-accent-hairline);
