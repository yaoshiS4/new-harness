Verbatim PrivilegeMatrix.module.css
/* ============================================================================
 * Guest privilege matrix — 4 privilege rows × 5 tier cols. Per-tier gem header,
 * green ✓ / grey lock cells, gold-tinted selected column, sticky metric column
 * on narrow screens. VIP tokens only.
 * ==========================================================================*/

.scroller {
  overflow-x: auto;
  border-radius: var(--vip-radius-table);
  border: 1px solid var(--vip-hairline);
  background: var(--vip-panel-bg);
  -webkit-overflow-scrolling: touch;
}

/* RULING B1 — `table-layout: fixed` so all FIVE tier columns are exactly the
 * same width. Under auto layout the widths tracked the tier NAME length and
 * came out 158/121/148/225/262px bronze→diamond, i.e. the top tier rendered
 * widest and the ladder read as a size ranking. Fixed layout removes that
 * entirely; rank is carried by column ORDER alone. */
.table {
  border-collapse: separate;
  border-spacing: 0;
  table-layout: fixed;
  width: 100%;
  min-width: 760px;
}

.srOnly {
  position: absolute;
  width: 1px;
  height: 1px;
  padding: 0;
  margin: -1px;
  overflow: hidden;
  clip: rect(0, 0, 0, 0);
  white-space: nowrap;
  border: 0;
}

.th,
.td {
  padding: var(--space-3);
  text-align: center;
  border-bottom: 1px solid var(--vip-hairline);
  color: var(--vip-text-secondary);
  vertical-align: middle;
}

/* Every header cell reserves the SAME 2px bottom rule, so marking the selected
 * column gold is a colour change with zero layout shift — and every tier column
 * keeps identical geometry (ruling B1: rank by position, never by size). */
thead .th {
  border-bottom: 2px solid var(--vip-hairline);
}

/* sticky first column (metric labels) for horizontal scroll (criterion 9) */
.rowHead {
  position: sticky;
  left: 0;
  z-index: 2;
  /* the ONE non-peer column; the five tier columns share the rest equally */
  width: 200px;
  text-align: left;
  color: var(--vip-text-primary);
  font-weight: var(--weight-semi);
  white-space: nowrap;
  background: var(--vip-panel-bg);
  border-right: 1px solid var(--vip-hairline);
}

thead .rowHead {
  z-index: 3;
  background: var(--vip-inner-bg);
}
