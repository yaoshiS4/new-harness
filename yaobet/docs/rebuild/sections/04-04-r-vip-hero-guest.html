Verbatim VipHeroGuest.module.css
/* ============================================================================
 * Luxury VIP — Guest hero. FLAT: a solid warm panel drawn by a load-bearing
 * gold hairline (ruling A2) with a full-bleed hero backdrop gradient — the ONE
 * gradient class the system still permits. NO shadow, NO glow, NO blur halo.
 * Two CTAs (register / login) on solid fills with a hover COLOUR shift.
 * Page-scoped dark palette (VIP tokens only).
 * ==========================================================================*/

.hero {
  position: relative;
  isolation: isolate;
  overflow: hidden;
  display: flex;
  flex-direction: column;
  align-items: center;
  text-align: center;
  gap: var(--space-3);
  padding: calc(var(--card-pad) * 1.75) var(--card-pad);
  border-radius: var(--radius-card);
  /* full-bleed hero backdrop — permitted as large background decoration */
  background:
    radial-gradient(
      120% 140% at 50% 0%,
      color-mix(in srgb, var(--vip-gold-mid) 10%, transparent) 0%,
      transparent 60%
    ),
    var(--vip-panel-bg);
  /* the border is the ONLY separation mechanism: shadows are banned and the
   * panel/page pair measures ~1.1:1 on its own (ruling A2). */
  border: 1px solid color-mix(in srgb, var(--vip-gold-mid) 34%, transparent);
}

.badge {
  display: inline-flex;
  align-items: center;
  gap: var(--space-1);
  padding: var(--space-1) var(--space-3);
  border-radius: var(--radius-pill);
  border: 1px solid color-mix(in srgb, var(--vip-gold-mid) 40%, transparent);
  background: color-mix(in srgb, var(--vip-gold-mid) 8%, transparent);
  color: var(--vip-gold-end);
  font-size: var(--t-micro);
  font-weight: var(--weight-semi);
  letter-spacing: 0.16em;
  text-transform: uppercase;
}

.badgeGem {
  width: 12px;
  height: 12px;
  border-radius: var(--vip-radius-gem);
  /* solid gold — the gem's glow was a shadow by another name */
  background: var(--vip-gold-mid);
  transform: rotate(45deg);
}

.title {
  font-family: var(--brand-font-display);
  font-weight: var(--weight-bold);
  font-size: clamp(var(--t-h1), 4.5vw, calc(var(--t-display) * 1.35));
  line-height: 1.12;
  letter-spacing: 0.01em;
  color: var(--vip-text-primary);
  max-width: 18ch;
  margin: 0;
}

.sub {
  color: var(--vip-text-secondary);
  font-size: var(--t-body);
  line-height: 1.6;
  max-width: 56ch;
  margin: 0;
}

.ctas {
  display: flex;
  gap: var(--space-3);
  flex-wrap: wrap;
  justify-content: center;
  margin-top: var(--space-2);
}

.ctaPrimary,
.ctaGhost {
  display: inline-flex;
  align-items: center;
  justify-content: center;
  min-height: var(--tap-min);
  padding: 0 calc(var(--space-4) * 1.5);
  border-radius: var(--vip-radius-control);
  font: inherit;
  font-weight: var(--weight-semi);
  letter-spacing: 0.02em;
  cursor: pointer;
  /* uniform button motion: scale(1.05) on hover, token duration/easing */
  transition:
    transform var(--dur-fast) var(--ease-standard),
    background var(--dur-fast) var(--ease-standard),
    border-color var(--dur-fast) var(--ease-standard),
    color var(--dur-fast) var(--ease-standard);
}

.ctaPrimary:hover,
.ctaGhost:hover {
  transform: scale(1.05);
}

/* SOLID gold fill — the 3-stop --vip-gold-grad is banned on a button. Ink is
 * --vip-gold-ink (dark) on --vip-gold-mid: white here would be an AA fail. */
.ctaPrimary {
  border: 0;
  background: var(--vip-gold-mid);
  color: var(--vip-gold-ink);
}

/* hover = a COLOUR shift, not a lift */
.ctaPrimary:hover {
  background: var(--vip-gold-end);
}

.ctaGhost {
  background: color-mix(in srgb, var(--vip-gold-mid) 4%, transparent);
  border: 1px solid color-mix(in srgb, var(--vip-gold-mid) 45%, transparent);
  color: var(--vip-gold-end);
}

.ctaGhost:hover {
  background: color-mix(in srgb, var(--vip-gold-mid) 12%, transparent);
  border-color: var(--vip-gold-mid);
}

/* BLUE MEANS FOCUS AND ONLY FOCUS — the locked neutral --focus-ring, never a
 * brand/tier colour (which would make focus indistinguishable from selection). */
.ctaPrimary:focus-visible,
.ctaGhost:focus-visible {
  outline: 2px solid var(--focus-ring);
  outline-offset: 2px;
}

@media (max-width: 900px) {
  .hero {
    padding: calc(var(--card-pad) * 1.25) var(--card-pad-sm);
  }
  .ctaPrimary,
  .ctaGhost {
    flex: 1 1 auto;
  }
}

@media (prefers-reduced-motion: reduce) {
  .ctaPrimary,
  .ctaGhost,
  .ctaPrimary:hover,
  .ctaGhost:hover {
    transform: none;
  }
}

6. States

Hover both: scale(1.05). Primary hover background #f6e39a. Ghost hover background rgba(227,193,90,.12), border-color #e3c15a. Focus-visible: 2px #6ea8ff outline offset 2px. No disabled/loading.

7. Motion

transform/background/border-color/color 200ms ease-standard; reduced motion → transform none.
